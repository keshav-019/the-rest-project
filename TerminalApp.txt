// Actions.tsx
import React, { useState } from 'react';
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Badge } from "@/components/ui/badge";
import { Dialog, DialogContent, DialogDescription, DialogHeader, DialogTitle, DialogTrigger } from "@/components/ui/dialog";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Textarea } from "@/components/ui/textarea";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { Plus, Workflow, Play, Pause, Settings, Trash2, Clock } from 'lucide-react';
import { useToast } from '@/hooks/useToast';

interface Action {
  id: string;
  name: string;
  description: string;
  commands: string[];
  targetServers: string[];
  schedule?: string;
  status: 'idle' | 'running' | 'completed' | 'failed';
  lastRun?: Date;
  environment: 'development' | 'staging' | 'production' | 'all';
}

export const Actions = () => {
  const { toast } = useToast();
  const [actions, setActions] = useState<Action[]>([
    {
      id: '1',
      name: 'Deploy to Production',
      description: 'Complete deployment pipeline from staging to production',
      commands: [
        'git pull origin main',
        'npm install',
        'npm run build',
        'pm2 restart all'
      ],
      targetServers: ['prod-server-1', 'prod-server-2'],
      status: 'completed',
      lastRun: new Date(Date.now() - 3600000),
      environment: 'production'
    },
    {
      id: '2',
      name: 'System Health Check',
      description: 'Run comprehensive system health checks across all servers',
      commands: [
        'df -h',
        'free -m',
        'systemctl status nginx',
        'docker ps'
      ],
      targetServers: ['all'],
      schedule: '0 */6 * * *',
      status: 'idle',
      environment: 'all'
    }
  ]);

  const [newAction, setNewAction] = useState({
    name: '',
    description: '',
    commands: '',
    targetServers: 'all',
    environment: 'development' as Action['environment']
  });

  const [isAddDialogOpen, setIsAddDialogOpen] = useState(false);

  const handleAddAction = () => {
    if (!newAction.name || !newAction.commands) {
      toast({
        title: "Error",
        description: "Please fill in all required fields",
        variant: "destructive",
      });
      return;
    }

    const action: Action = {
      id: Date.now().toString(),
      name: newAction.name,
      description: newAction.description,
      commands: newAction.commands.split('\n').filter(cmd => cmd.trim()),
      targetServers: newAction.targetServers === 'all' ? ['all'] : [newAction.targetServers],
      status: 'idle',
      environment: newAction.environment
    };

    setActions([...actions, action]);
    setNewAction({ name: '', description: '', commands: '', targetServers: 'all', environment: 'development' });
    setIsAddDialogOpen(false);

    toast({
      title: "Success",
      description: `Action "${action.name}" created successfully`,
    });
  };

  const executeAction = (actionId: string) => {
    setActions(actions.map(a =>
      a.id === actionId
        ? { ...a, status: 'running' }
        : a
    ));

    // Simulate action execution
    setTimeout(() => {
      setActions(actions.map(a =>
        a.id === actionId
          ? { ...a, status: 'completed', lastRun: new Date() }
          : a
      ));
      toast({
        title: "Action Completed",
        description: "Action executed successfully",
      });
    }, 3000);
  };

  const handleDeleteAction = (actionId: string) => {
    setActions(actions.filter(a => a.id !== actionId));
    toast({
      title: "Action Deleted",
      description: "Action has been removed from your workflows",
    });
  };

  const getStatusColor = (status: Action['status']) => {
    switch (status) {
      case 'running': return 'bg-blue-500';
      case 'completed': return 'bg-green-500';
      case 'failed': return 'bg-red-500';
      case 'idle': return 'bg-gray-500';
    }
  };

  const getEnvironmentColor = (environment: Action['environment']) => {
    switch (environment) {
      case 'development': return 'bg-green-500/10 text-green-500 border-green-500/20';
      case 'staging': return 'bg-yellow-500/10 text-yellow-500 border-yellow-500/20';
      case 'production': return 'bg-red-500/10 text-red-500 border-red-500/20';
      case 'all': return 'bg-blue-500/10 text-blue-500 border-blue-500/20';
    }
  };

  const getStatusText = (status: Action['status']) => {
    switch (status) {
      case 'running': return 'Running';
      case 'completed': return 'Completed';
      case 'failed': return 'Failed';
      case 'idle': return 'Ready';
    }
  };

  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between">
        <div>
          <h2 className="text-2xl font-bold tracking-tight">Action Workflows</h2>
          <p className="text-muted-foreground">
            Automate deployment and maintenance tasks across your infrastructure
          </p>
        </div>
        <Dialog open={isAddDialogOpen} onOpenChange={setIsAddDialogOpen}>
          <DialogTrigger asChild>
            <Button>
              <Plus className="h-4 w-4 mr-2" />
              Create Action
            </Button>
          </DialogTrigger>
          <DialogContent className="sm:max-w-[500px]">
            <DialogHeader>
              <DialogTitle>Create New Action</DialogTitle>
              <DialogDescription>
                Define a new automated workflow for your servers
              </DialogDescription>
            </DialogHeader>
            <div className="grid gap-4 py-4">
              <div className="grid gap-2">
                <Label htmlFor="actionName">Action Name</Label>
                <Input
                  id="actionName"
                  value={newAction.name}
                  onChange={(e) => setNewAction({...newAction, name: e.target.value})}
                  placeholder="Deploy Application"
                />
              </div>
              <div className="grid gap-2">
                <Label htmlFor="actionDescription">Description</Label>
                <Input
                  id="actionDescription"
                  value={newAction.description}
                  onChange={(e) => setNewAction({...newAction, description: e.target.value})}
                  placeholder="What this action does"
                />
              </div>
              <div className="grid gap-2">
                <Label htmlFor="commands">Commands (one per line)</Label>
                <Textarea
                  id="commands"
                  value={newAction.commands}
                  onChange={(e) => setNewAction({...newAction, commands: e.target.value})}
                  placeholder="git pull origin main&#10;npm install&#10;npm run build"
                  rows={6}
                />
              </div>
              <div className="grid gap-2">
                <Label htmlFor="environment">Environment</Label>
                <Select value={newAction.environment} onValueChange={(value: Action['environment']) => setNewAction({...newAction, environment: value})}>
                  <SelectTrigger>
                    <SelectValue />
                  </SelectTrigger>
                  <SelectContent>
                    <SelectItem value="development">Development</SelectItem>
                    <SelectItem value="staging">Staging</SelectItem>
                    <SelectItem value="production">Production</SelectItem>
                    <SelectItem value="all">All Environments</SelectItem>
                  </SelectContent>
                </Select>
              </div>
            </div>
            <div className="flex justify-end space-x-2">
              <Button variant="outline" onClick={() => setIsAddDialogOpen(false)}>
                Cancel
              </Button>
              <Button onClick={handleAddAction}>
                Create Action
              </Button>
            </div>
          </DialogContent>
        </Dialog>
      </div>

      <div className="grid gap-4 md:grid-cols-2 lg:grid-cols-3">
        {actions.map((action) => (
          <Card key={action.id} className="relative">
            <CardHeader className="pb-3">
              <div className="flex items-center justify-between">
                <div className="flex items-center space-x-2">
                  <Workflow className="h-5 w-5 text-primary" />
                  <CardTitle className="text-lg">{action.name}</CardTitle>
                </div>
                <div className="flex items-center space-x-1">
                  <div className={`h-2 w-2 rounded-full ${getStatusColor(action.status)}`} />
                  <Badge variant="secondary" className="text-xs">
                    {getStatusText(action.status)}
                  </Badge>
                </div>
              </div>
              <CardDescription>
                {action.description}
              </CardDescription>
            </CardHeader>
            <CardContent>
              <div className="space-y-3">
                <div className="text-sm text-muted-foreground">
                  <div className="flex items-center gap-2 mb-2">
                    <Badge variant="secondary" className={`text-xs border ${getEnvironmentColor(action.environment)}`}>
                      {action.environment}
                    </Badge>
                    {action.schedule && (
                      <Badge variant="secondary" className="text-xs bg-purple-500/10 text-purple-500 border-purple-500/20">
                        <Clock className="h-3 w-3 mr-1" />
                        Scheduled
                      </Badge>
                    )}
                  </div>
                  <p>Commands: {action.commands.length}</p>
                  <p>Targets: {action.targetServers.join(', ')}</p>
                  {action.lastRun && (
                    <p>Last run: {action.lastRun.toLocaleString()}</p>
                  )}
                </div>
                <div className="flex space-x-2">
                  <Button
                    size="sm"
                    onClick={() => executeAction(action.id)}
                    disabled={action.status === 'running'}
                    className="flex-1"
                  >
                    {action.status === 'running' ? (
                      <>
                        <Pause className="h-3 w-3 mr-1" />
                        Running...
                      </>
                    ) : (
                      <>
                        <Play className="h-3 w-3 mr-1" />
                        Execute
                      </>
                    )}
                  </Button>
                  <Button size="sm" variant="outline">
                    <Settings className="h-3 w-3" />
                  </Button>
                  <Button
                    size="sm"
                    variant="outline"
                    onClick={() => handleDeleteAction(action.id)}
                  >
                    <Trash2 className="h-3 w-3" />
                  </Button>
                </div>
              </div>
            </CardContent>
          </Card>
        ))}
      </div>
    </div>
  );
};

// KeyManagement.tsx


import React, { useState } from 'react';
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Badge } from "@/components/ui/badge";
import { Dialog, DialogContent, DialogDescription, DialogHeader, DialogTitle, DialogTrigger } from "@/components/ui/dialog";
import { Input } from "@/components/ui/input";
import { Label } from "@/components/ui/label";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { Textarea } from "@/components/ui/textarea";
import { Plus, Key, Copy, Trash2, Download } from 'lucide-react';
import { useToast } from '@/hooks/useToast';

interface SSHKey {
    id: string;
    name: string;
    type: 'rsa' | 'ed25519' | 'ecdsa';
    fingerprint: string;
    createdAt: Date;
    lastUsed?: Date;
    isProtected: boolean;
}

export const KeyManagement = () => {
    const { toast } = useToast();
    const [keys, setKeys] = useState<SSHKey[]>([
        {
            id: '1',
            name: 'Production Deploy Key',
            type: 'rsa',
            fingerprint: 'SHA256:nThbg6kXUpJWGl7E1IGOCspRomTxdCARLviKw6E5SY8',
            createdAt: new Date(Date.now() - 86400000),
            lastUsed: new Date(Date.now() - 3600000),
            isProtected: true
        },
        {
            id: '2',
            name: 'Development Key',
            type: 'ed25519',
            fingerprint: 'SHA256:1234567890abcdef1234567890abcdef12345678',
            createdAt: new Date(Date.now() - 172800000),
            isProtected: false
        }
    ]);

    const [newKey, setNewKey] = useState({
        name: '',
        type: 'rsa' as 'rsa' | 'ed25519' | 'ecdsa',
        keyContent: '',
        passphrase: ''
    });

    const [isAddDialogOpen, setIsAddDialogOpen] = useState(false);

    const handleAddKey = () => {
        if (!newKey.name || !newKey.keyContent) {
            toast({
                title: "Error",
                description: "Please fill in all required fields",
                variant: "destructive",
            });
            return;
        }

        const key: SSHKey = {
            id: Date.now().toString(),
            name: newKey.name,
            type: newKey.type,
            fingerprint: `SHA256:${Math.random().toString(36).substring(2, 45)}`,
            createdAt: new Date(),
            isProtected: newKey.passphrase.length > 0
        };

        setKeys([...keys, key]);
        setNewKey({ name: '', type: 'rsa', keyContent: '', passphrase: '' });
        setIsAddDialogOpen(false);

        toast({
            title: "Success",
            description: `SSH key "${key.name}" added successfully`,
        });
    };

    const handleDeleteKey = (keyId: string) => {
        setKeys(keys.filter(k => k.id !== keyId));
        toast({
            title: "Key Deleted",
            description: "SSH key has been removed from your keychain",
        });
    };

    const copyFingerprint = (fingerprint: string) => {
        navigator.clipboard.writeText(fingerprint);
        toast({
            title: "Copied",
            description: "Fingerprint copied to clipboard",
        });
    };

    const getTypeColor = (type: SSHKey['type']) => {
        switch (type) {
            case 'rsa': return 'bg-blue-500/10 text-blue-500 border-blue-500/20';
            case 'ed25519': return 'bg-green-500/10 text-green-500 border-green-500/20';
            case 'ecdsa': return 'bg-purple-500/10 text-purple-500 border-purple-500/20';
        }
    };

    return (
        <div className="space-y-6">
            <div className="flex items-center justify-between">
                <div>
                    <h2 className="text-2xl font-bold tracking-tight">SSH Key Management</h2>
                    <p className="text-muted-foreground">
                        Manage your SSH keys and configure secure authentication
                    </p>
                </div>
                <Dialog open={isAddDialogOpen} onOpenChange={setIsAddDialogOpen}>
                    <DialogTrigger asChild>
                        <Button>
                            <Plus className="h-4 w-4 mr-2" />
                            Add SSH Key
                        </Button>
                    </DialogTrigger>
                    <DialogContent className="sm:max-w-[500px]">
                        <DialogHeader>
                            <DialogTitle>Add New SSH Key</DialogTitle>
                            <DialogDescription>
                                Add an existing SSH key or generate a new one
                            </DialogDescription>
                        </DialogHeader>
                        <div className="grid gap-4 py-4">
                            <div className="grid gap-2">
                                <Label htmlFor="keyName">Key Name</Label>
                                <Input
                                    id="keyName"
                                    value={newKey.name}
                                    onChange={(e) => setNewKey({ ...newKey, name: e.target.value })}
                                    placeholder="My SSH Key"
                                />
                            </div>
                            <div className="grid gap-2">
                                <Label htmlFor="keyType">Key Type</Label>
                                <Select value={newKey.type} onValueChange={(value: 'rsa' | 'ed25519' | 'ecdsa') => setNewKey({ ...newKey, type: value })}>
                                    <SelectTrigger>
                                        <SelectValue />
                                    </SelectTrigger>
                                    <SelectContent>
                                        <SelectItem value="rsa">RSA</SelectItem>
                                        <SelectItem value="ed25519">ED25519</SelectItem>
                                        <SelectItem value="ecdsa">ECDSA</SelectItem>
                                    </SelectContent>
                                </Select>
                            </div>
                            <div className="grid gap-2">
                                <Label htmlFor="keyContent">Private Key Content</Label>
                                <Textarea
                                    id="keyContent"
                                    value={newKey.keyContent}
                                    onChange={(e) => setNewKey({ ...newKey, keyContent: e.target.value })}
                                    placeholder="-----BEGIN OPENSSH PRIVATE KEY-----..."
                                    rows={8}
                                />
                            </div>
                            <div className="grid gap-2">
                                <Label htmlFor="passphrase">Passphrase (Optional)</Label>
                                <Input
                                    id="passphrase"
                                    type="password"
                                    value={newKey.passphrase}
                                    onChange={(e) => setNewKey({ ...newKey, passphrase: e.target.value })}
                                    placeholder="Enter passphrase if key is protected"
                                />
                            </div>
                        </div>
                        <div className="flex justify-end space-x-2">
                            <Button variant="outline" onClick={() => setIsAddDialogOpen(false)}>
                                Cancel
                            </Button>
                            <Button onClick={handleAddKey}>
                                Add Key
                            </Button>
                        </div>
                    </DialogContent>
                </Dialog>
            </div>

            <div className="grid gap-4 md:grid-cols-2 lg:grid-cols-3">
                {keys.map((key) => (
                    <Card key={key.id} className="relative">
                        <CardHeader className="pb-3">
                            <div className="flex items-center justify-between">
                                <div className="flex items-center space-x-2">
                                    <Key className="h-5 w-5 text-primary" />
                                    <CardTitle className="text-lg">{key.name}</CardTitle>
                                </div>
                                <div className="flex items-center space-x-2">
                                    <Badge variant="secondary" className={`text-xs border ${getTypeColor(key.type)}`}>
                                        {key.type.toUpperCase()}
                                    </Badge>
                                    {key.isProtected && (
                                        <Badge variant="secondary" className="text-xs bg-amber-500/10 text-amber-500 border-amber-500/20">
                                            Protected
                                        </Badge>
                                    )}
                                </div>
                            </div>
                            <CardDescription>
                                Created: {key.createdAt.toLocaleDateString()}
                            </CardDescription>
                        </CardHeader>
                        <CardContent>
                            <div className="space-y-3">
                                <div className="text-sm text-muted-foreground">
                                    <p className="font-mono text-xs break-all">{key.fingerprint}</p>
                                    {key.lastUsed && (
                                        <p className="mt-1">Last used: {key.lastUsed.toLocaleString()}</p>
                                    )}
                                </div>
                                <div className="flex space-x-2">
                                    <Button
                                        size="sm"
                                        variant="outline"
                                        onClick={() => copyFingerprint(key.fingerprint)}
                                        className="flex-1"
                                    >
                                        <Copy className="h-3 w-3 mr-1" />
                                        Copy
                                    </Button>
                                    <Button size="sm" variant="outline">
                                        <Download className="h-3 w-3" />
                                    </Button>
                                    <Button
                                        size="sm"
                                        variant="outline"
                                        onClick={() => handleDeleteKey(key.id)}
                                    >
                                        <Trash2 className="h-3 w-3" />
                                    </Button>
                                </div>
                            </div>
                        </CardContent>
                    </Card>
                ))}
            </div>
        </div>
    );
};

// LocalTerminal.tsx


import React, { useEffect, useRef, useState } from 'react';
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { X, Minimize2 } from 'lucide-react';

interface LocalTerminalProps {
    onClose: () => void;
}

export const LocalTerminal = ({ onClose }: LocalTerminalProps) => {
    const terminalRef = useRef<HTMLDivElement>(null);
    const [input, setInput] = useState('');
    const [history, setHistory] = useState<string[]>([]);
    const [currentDirectory, setCurrentDirectory] = useState('~');
    const [isMinimized, setIsMinimized] = useState(false);

    const executeCommand = (command: string) => {
        const parts = command.trim().split(' ');
        const cmd = parts[0];
        const args = parts.slice(1);

        let output = '';

        switch (cmd) {
            case 'ls':
                output = 'Documents  Downloads  Desktop  Pictures  Music  Videos  Projects';
                break;
            case 'pwd':
                output = `/home/user/${currentDirectory}`;
                break;
            case 'whoami':
                output = 'keshav';
                break;
            case 'date':
                output = new Date().toString();
                break;
            case 'echo':
                output = args.join(' ');
                break;
            case 'cd':
                if (args.length === 0) {
                    setCurrentDirectory('~');
                    output = '';
                } else {
                    const newDir = args[0];
                    if (newDir === '..') {
                        const pathParts = currentDirectory.split('/');
                        pathParts.pop();
                        setCurrentDirectory(pathParts.join('/') || '~');
                    } else {
                        setCurrentDirectory(currentDirectory === '~' ? newDir : `${currentDirectory}/${newDir}`);
                    }
                    output = '';
                }
                break;
            case 'clear':
                setHistory([]);
                return;
            case 'help':
                output = `Available commands:
ls       - List directory contents
pwd      - Print working directory
whoami   - Print current user
date     - Print current date
echo     - Echo text
cd       - Change directory
clear    - Clear terminal
mkdir    - Create directory
touch    - Create file
cat      - Display file contents
ps       - Show running processes
uname    - System information`;
                break;
            case 'mkdir':
                output = args.length > 0 ? `Directory '${args[0]}' created` : 'mkdir: missing operand';
                break;
            case 'touch':
                output = args.length > 0 ? `File '${args[0]}' created` : 'touch: missing operand';
                break;
            case 'cat':
                output = args.length > 0 ? `Contents of ${args[0]}: Hello World!` : 'cat: missing operand';
                break;
            case 'ps':
                output = `PID    COMMAND
1234   bash
5678   node
9012   code
3456   chrome`;
                break;
            case 'uname':
                output = 'Linux terminal-emulator 5.15.0 x86_64';
                break;
            default:
                output = `Command '${cmd}' not found. Type 'help' for available commands.`;
        }

        return output;
    };

    const handleInputSubmit = (e: React.KeyboardEvent) => {
        if (e.key === 'Enter') {
            const command = input.trim();
            if (command) {
                const prompt = `user@terminal:${currentDirectory}$ ${command}`;
                const output = executeCommand(command);

                setHistory(prev => {
                    const newHistory = [...prev, prompt];
                    if (output) {
                        newHistory.push(output);
                    }
                    return newHistory;
                });

                setInput('');
            }
        }
    };

    useEffect(() => {
        if (terminalRef.current) {
            terminalRef.current.scrollTop = terminalRef.current.scrollHeight;
        }
    }, [history]);

    if (isMinimized) {
        return (
            <div className="fixed bottom-4 left-4 z-50">
                <Button
                    onClick={() => setIsMinimized(false)}
                    className="bg-zinc-800 hover:bg-zinc-700 text-white"
                >
                    Terminal
                </Button>
            </div>
        );
    }

    return (
        <div className="fixed inset-4 z-50 bg-black/50 backdrop-blur-sm flex items-center justify-center">
            <Card className="w-full max-w-4xl h-96 bg-zinc-900 border-zinc-700 text-green-400 font-mono">
                <CardHeader className="pb-2">
                    <div className="flex items-center justify-between">
                        <CardTitle className="text-sm text-zinc-300">
                            Local Terminal
                        </CardTitle>
                        <div className="flex gap-2">
                            <Button
                                size="sm"
                                variant="ghost"
                                onClick={() => setIsMinimized(true)}
                                className="h-6 w-6 p-0 text-zinc-400 hover:text-white hover:bg-zinc-700"
                            >
                                <Minimize2 className="h-3 w-3" />
                            </Button>
                            <Button
                                size="sm"
                                variant="ghost"
                                onClick={onClose}
                                className="h-6 w-6 p-0 text-zinc-400 hover:text-white hover:bg-zinc-700"
                            >
                                <X className="h-3 w-3" />
                            </Button>
                        </div>
                    </div>
                </CardHeader>
                <CardContent className="flex flex-col h-full pb-4">
                    <div
                        ref={terminalRef}
                        className="flex-1 overflow-y-auto bg-black p-3 rounded text-sm leading-relaxed"
                    >
                        <div className="text-green-400 mb-2">
                            Welcome to Local Terminal Emulator v1.0
                        </div>
                        <div className="text-green-400 mb-2">
                            Type 'help' for available commands.
                        </div>
                        {history.map((line, index) => (
                            <div key={index} className="whitespace-pre-wrap">
                                {line}
                            </div>
                        ))}
                        <div className="flex items-center">
                            <span className="text-green-400">user@terminal:{currentDirectory}$ </span>
                            <input
                                title='terminal_command'
                                type="text"
                                value={input}
                                onChange={(e) => setInput(e.target.value)}
                                onKeyDown={handleInputSubmit}
                                className="flex-1 bg-transparent outline-none text-green-400 ml-1"
                                autoFocus
                                placeholder=""
                            />
                        </div>
                    </div>
                </CardContent>
            </Card>
        </div>
    );
};

import { Terminal } from './Terminal';
import { Plus, Server, Play, Settings, Trash2 } from 'lucide-react';
import { useToast } from '@/hooks/useToast';

interface Server {
    id: string;
    name: string;
    host: string;
    port: number;
    username: string;
    authType: 'password' | 'key';
    password?: string;
    keyId?: string;
    status: 'connected' | 'disconnected' | 'connecting';
    lastConnected?: Date;
}

            id: '1',
            name: 'Production Server',
            host: '192.168.1.100',
            port: 22,
            username: 'admin',

            keyId: 'prod-key',
            status: 'disconnected',
            lastConnected: new Date(Date.now() - 3600000)
        },

            id: '2',
            name: 'Development Server',
            host: '10.0.0.50',

            status: 'connected',

                title: "Error",
                description: "Please fill in all required fields",
                variant: "destructive",
            });
            return;

            id: Date.now().toString(),
            name: newServer.name,
            host: newServer.host,
            port: newServer.port,
            username: newServer.username,
            authType: newServer.authType,
            password: newServer.authType === 'password' ? newServer.password : undefined,
            keyId: newServer.authType === 'key' ? newServer.keyId : undefined,
            status: 'disconnected'
        };

        setServers([...servers, server]);
        setNewServer({
            name: '',
            host: '',
            port: 22,
            username: '',
            authType: 'password',
            password: '',
            keyId: ''
        });
        setIsAddDialogOpen(false);

        toast({
            title: "Success",
            description: `Server "${server.name}" added successfully`,
        });
    };

    const handleConnect = (server: Server) => {
        setServers(servers.map(s =>
            s.id === server.id
                ? { ...s, status: 'connecting' }
                : s
        ));

        // Simulate connection delay
        setTimeout(() => {
            setServers(servers.map(s =>
                s.id === server.id

                    : s
            ));
            setActiveTerminal(server);
            toast({

                description: `Connected to ${server.name}`,

        }, 1500);

        setServers(servers.filter(s => s.id !== serverId));

            title: "Server Deleted",
            description: "Server has been removed from your list",
        });
    };

    const getStatusColor = (status: Server['status']) => {
        switch (status) {

            case 'connecting': return 'bg-yellow-500';
            case 'disconnected': return 'bg-gray-500';

    const getStatusText = (status: Server['status']) => {
        switch (status) {

            case 'connecting': return 'Connecting...';
            case 'disconnected': return 'Disconnected';
        }
    };

    return (
        <div className="space-y-6">
            <div className="flex items-center justify-between">
                <div>
                    <h2 className="text-2xl font-bold tracking-tight">Server Management</h2>
                    <p className="text-muted-foreground">

                            Add Server

                                    id="name"

                                    onChange={(e) => setNewServer({ ...newServer, name: e.target.value })}
                                    placeholder="Production Server"
                                />
                            </div>
                            <div className="grid gap-2">
                                <Label htmlFor="host">Host</Label>
                                <Input
                                    id="host"
                                    value={newServer.host}
                                    onChange={(e) => setNewServer({ ...newServer, host: e.target.value })}
                                    placeholder="192.168.1.100"
                                />
                            </div>
                            <div className="grid gap-2">
                                <Label htmlFor="port">Port</Label>
                                <Input
                                    id="port"
                                    type="number"
                                    value={newServer.port}
                                    onChange={(e) => setNewServer({ ...newServer, port: parseInt(e.target.value) })}
                                    placeholder="22"
                                />
                            </div>
                            <div className="grid gap-2">
                                <Label htmlFor="username">Username</Label>
                                <Input
                                    id="username"
                                    value={newServer.username}
                                    onChange={(e) => setNewServer({ ...newServer, username: e.target.value })}
                                    placeholder="admin"
                                />
                            </div>
                            <div className="grid gap-2">
                                <Label htmlFor="authType">Authentication Type</Label>
                                <Select value={newServer.authType} onValueChange={(value: 'password' | 'key') => setNewServer({ ...newServer, authType: value })}>
                                    <SelectTrigger>
                                        <SelectValue />
                                    </SelectTrigger>
                                    <SelectContent>
                                        <SelectItem value="password">Password</SelectItem>
                                        <SelectItem value="key">SSH Key</SelectItem>
                                    </SelectContent>
                                </Select>
                            </div>
                            {newServer.authType === 'password' && (
                                <div className="grid gap-2">
                                    <Label htmlFor="password">Password</Label>
                                    <Input
                                        id="password"
                                        type="password"
                                        value={newServer.password}
                                        onChange={(e) => setNewServer({ ...newServer, password: e.target.value })}
                                        placeholder="Enter password"
                                    />
                                </div>
                            )}
                            {newServer.authType === 'key' && (
                                <div className="grid gap-2">
                                    <Label htmlFor="keyId">SSH Key</Label>
                                    <Select value={newServer.keyId} onValueChange={(value) => setNewServer({ ...newServer, keyId: value })}>
                                        <SelectTrigger>
                                            <SelectValue placeholder="Select SSH key" />
                                        </SelectTrigger>
                                        <SelectContent>
                                            <SelectItem value="prod-key">Production Key</SelectItem>
                                            <SelectItem value="dev-key">Development Key</SelectItem>
                                        </SelectContent>
                                    </Select>
                                </div>
                            )}
                        </div>
                        <div className="flex justify-end space-x-2">
                            <Button variant="outline" onClick={() => setIsAddDialogOpen(false)}>
                                Cancel
                            </Button>
                            <Button onClick={handleAddServer}>
                                Add Server
                            </Button>
                        </div>
                    </DialogContent>
                </Dialog>
            </div>

            <div className="grid gap-4 md:grid-cols-2 lg:grid-cols-3">
                {servers.map((server) => (

                        <CardHeader className="pb-3">

                                </div>
                            </div>
                            <CardDescription>
                                {server.username}@{server.host}:{server.port}
                            </CardDescription>
                        </CardHeader>
                        <CardContent>
                            <div className="space-y-3">
                                <div className="text-sm text-muted-foreground">
                                    <p>Auth: {server.authType === 'key' ? `SSH Key (${server.keyId})` : 'Password'}</p>
                                    {server.lastConnected && (
                                        <p>Last connected: {server.lastConnected.toLocaleString()}</p>
                                    )}
                                </div>
                                <div className="flex space-x-2">
                                    <Button
                                        size="sm"
                                        onClick={() => handleConnect(server)}
                                        disabled={server.status === 'connecting'}
                                        className="flex-1"
                                    >
                                        <Play className="h-3 w-3 mr-1" />
                                        {server.status === 'connecting' ? 'Connecting...' : 'Connect'}
                                    </Button>
                                    <Button size="sm" variant="outline">
                                        <Settings className="h-3 w-3" />
                                    </Button>
                                    <Button
                                        size="sm"
                                        variant="outline"
                                        onClick={() => handleDeleteServer(server.id)}
                                    >
                                        <Trash2 className="h-3 w-3" />
                                    </Button>
                                </div>
                            </div>
                        </CardContent>
                    </Card>
                ))}
            </div>

            {activeTerminal && (
                <Terminal
                    server={activeTerminal}
                    onClose={() => setActiveTerminal(null)}
                />
            )}
        </div>
    );
};

        }

                        </DialogHeader>
                        <div className="grid gap-4 py-4">
                            <div className="grid gap-2">

                            </div>

// Terminal.tsx


import React, { useEffect, useRef, useState } from 'react';
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { X, Minimize2 } from 'lucide-react';

interface TerminalProps {
    server: {
        id: string;
        name: string;
        host: string;
        port: number;
        username: string;
    };
    onClose: () => void;
}

export const Terminal = ({ server, onClose }: TerminalProps) => {
    const terminalRef = useRef<HTMLDivElement>(null);
    const [input, setInput] = useState('');
    const [output, setOutput] = useState<string[]>([
        `Connecting to ${server.username}@${server.host}:${server.port}...`,
        `Connected to ${server.name}`,
        `Welcome to ${server.name}`,
        `Last login: ${new Date().toLocaleString()}`,
        `${server.username}@${server.host}:~$`
    ]);
    const [isMinimized, setIsMinimized] = useState(false);

    const handleInputSubmit = (e: React.KeyboardEvent) => {
        if (e.key === 'Enter') {
            const command = input.trim();
            if (command) {
                setOutput(prev => [
                    ...prev,
                    `${server.username}@${server.host}:~$ ${command}`,
                    `Command executed: ${command}`,
                    `${server.username}@${server.host}:~$`
                ]);
                setInput('');
            }
        }
    };

    useEffect(() => {
        if (terminalRef.current) {
            terminalRef.current.scrollTop = terminalRef.current.scrollHeight;
        }
    }, [output]);

    if (isMinimized) {
        return (
            <div className="fixed bottom-4 left-4 z-50">
                <Button
                    onClick={() => setIsMinimized(false)}
                    className="bg-zinc-800 hover:bg-zinc-700 text-white"
                >
                    Terminal - {server.name}
                </Button>
            </div>
        );
    }

    return (
        <div className="fixed inset-4 z-50 bg-black/50 backdrop-blur-sm flex items-center justify-center">
            <Card className="w-full max-w-4xl h-96 bg-zinc-900 border-zinc-700 text-green-400 font-mono">
                <CardHeader className="pb-2">
                    <div className="flex items-center justify-between">
                        <CardTitle className="text-sm text-zinc-300">
                            Terminal - {server.name} ({server.host}:{server.port})
                        </CardTitle>
                        <div className="flex gap-2">
                            <Button
                                size="sm"
                                variant="ghost"
                                onClick={() => setIsMinimized(true)}
                                className="h-6 w-6 p-0 text-zinc-400 hover:text-white hover:bg-zinc-700"
                            >
                                <Minimize2 className="h-3 w-3" />
                            </Button>
                            <Button
                                size="sm"
                                variant="ghost"
                                onClick={onClose}
                                className="h-6 w-6 p-0 text-zinc-400 hover:text-white hover:bg-zinc-700"
                            >
                                <X className="h-3 w-3" />
                            </Button>
                        </div>
                    </div>
                </CardHeader>
                <CardContent className="flex flex-col h-full pb-4">
                    <div
                        ref={terminalRef}
                        className="flex-1 overflow-y-auto bg-black p-3 rounded text-sm leading-relaxed"
                    >
                        {output.map((line, index) => (
                            <div key={index} className="whitespace-pre-wrap">
                                {line}
                            </div>
                        ))}
                        <div className="flex items-center">
                            <span className="text-green-400">{server.username}@{server.host}:~$ </span>
                            <input
                                type="text"
                                value={input}
                                onChange={(e) => setInput(e.target.value)}
                                onKeyDown={handleInputSubmit}
                                className="flex-1 bg-transparent outline-none text-green-400 ml-1"
                                autoFocus
                                placeholder="Type your command..."
                            />
                        </div>
                    </div>
                </CardContent>
            </Card>
        </div>
    );
};

'use client'
import React from 'react';

import UserDropdown from './UserDropdown';
import { PlusIcon, SearchIcon } from './Icons';
import { Environment, Team } from '@/types/User';
import EnvironmentSelector from '../Environment/EnvironmentSelector';

interface HeaderProps {
    onAddCollection: (value: string) => void,
    toSearch: boolean,
    parentComponent: string,
    environments: Environment[] | undefined,
    initials: string,
    username: string,
    email: string,
    displayName: string,
    autoSave: boolean,
    setAutoSave: (value: boolean) => void,
    teamMode: Team | null,
    teams: Team[],
    onTeamSelect: (team: Team | null) => void;
    onExitTeamMode: () => void;
    showTeams: boolean;
    setShowTeams: (value: boolean) => void;
    activeEnvironmentId: string | null;
    onEnvironmentSelect: (environmentId: string) => void;
}

const HeaderComponent = ({ toSearch, parentComponent, onAddCollection, environments, initials, username, email, displayName, autoSave, setAutoSave, teamMode, teams, onTeamSelect, onExitTeamMode, showTeams, setShowTeams, activeEnvironmentId, onEnvironmentSelect}: HeaderProps) => {
    return (
        <header className="flex items-center justify-between h-16 bg-white dark:bg-gray-800 border-b border-gray-200 dark:border-gray-700 px-6">
        <div className="flex items-center space-x-8 w-full">
        <div className="w-50"> {/* Matches CollectionsTree width */}
        <LogoDisplay />
        </div>

        {toSearch &&
            (
                <div className="flex-1 max-w-2xl mx-4"> {/* Centered search */}
                <div className="relative">
                <input
                type="text"
                placeholder={parentComponent === 'Environment' ? "Search environments..." : "Search collections..."}
                className="w-full pl-10 pr-4 py-2 rounded-lg border border-gray-300 dark:border-gray-600 bg-gray-50 dark:bg-gray-700 text-gray-900 dark:text-gray-100 focus:outline-none focus:ring-2 focus:ring-blue-500"
                />
                <div className="absolute left-3 top-2.5 text-gray-400">
                <SearchIcon />
                </div>
                </div>
                </div>
            )
        }
        {!toSearch &&
            (
                <div></div>
            )
        }
        </div>

        <div className="flex items-center space-x-4">
        {parentComponent === 'Collections' &&
            <button
            className="px-4 py-2 bg-blue-600 hover:bg-blue-700 text-white font-medium rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500 focus:ring-offset-2 transition-colors cursor-pointer flex items-center w-48"
            onClick={(e) => {
                e.preventDefault();
                onAddCollection('New Collection');
            }}
            >
            <PlusIcon />
            <span className="ml-2">New Collection</span>
            </button>
        }

        {parentComponent === 'Request Builder' &&
            <EnvironmentSelector
            environments={environments}
            activeEnvironmentId={activeEnvironmentId}
            onEnvironmentSelect={onEnvironmentSelect}
            disabled={!environments || environments.length === 0}
            />
        }

        <UserDropdown
        username={username}
        displayName={displayName}
        email={email}
        initials={initials}
        teams={teams}
        autoSave={autoSave}
        setAutoSave={setAutoSave}
        teamMode={teamMode}
        onExitTeamMode={onExitTeamMode}
        setTeamMode={onTeamSelect}
        setShowTeams={setShowTeams}
        showTeams={showTeams}
        />
        </div>
        </header>
    );
};

export default HeaderComponent;
